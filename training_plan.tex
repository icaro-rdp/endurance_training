% Watts at given % FTP: \W{90} -> "230"
\newcommand{\W}[1]{\fpeval{round(\currentFTP*#1/100,0)}}

% Single % FTP with watts: \sgl{106} -> "106% (270W)"
\newcommand{\sgl}[1]{#1\%~{\scriptsize(\W{#1}W)}}

% Range % FTP with watts: \rng{90}{94} -> "90-94% (230-240W)"
\newcommand{\rng}[2]{#1--#2\%~{\scriptsize(\W{#1}--\W{#2}W)}}

% Run pace at % of CS speed: \rpace{82} -> "4:53/km"
% Round-then-split avoids "m:60" edge cases from float rounding.
% NOTE: xfp in TeX Live 2022 has no mod(); we expand it inline as
%       x - 60*floor(x/60), which is exact for integer total seconds.
\newcommand{\rpace}[1]{%
  \fpeval{floor(round(\currentCS/(#1/100),0)/60)}:%
  \ifnum\fpeval{round(\currentCS/(#1/100),0)-60*floor(round(\currentCS/(#1/100),0)/60)}<10 0\fi%
  \fpeval{round(\currentCS/(#1/100),0)-60*floor(round(\currentCS/(#1/100),0)/60)}/km%
}

\section*{Phase 3 --- VO\textsubscript{2}Max: Block Periodization (Weeks 9--12)}

\noindent\emph{\footnotesize W9 is a 5-session HIT block: no long ride, running cut to one easy 6\,km.
W10--11 carry one HIT/week with full LIT volume returned --- this is when the delayed adaptation surfaces.
W12 recovery captures the supercompensation; retest at the END of W12, not the start.
Interval format: alternating 6x5' and 5x6', all-out for highest mean power, 2.5'/3' recovery.
Source: Rønnestad, Hansen \& Ellefsen, Scand J Med Sci Sports (2012).}
